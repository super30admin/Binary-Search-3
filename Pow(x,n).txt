class Solution {
    public double myPow(double x, int n) {
        double result = 1;
        if(n < 0) x = 1/x;
        return myPow(x,n, result) ;
    }
    private double myPow(double x, int n,double result){
        // base
        if(n == 0) return 1;
                    
        //logic 
        double re = myPow(x,n/2,result);
        if(n %2  != 0 ){
            result = re * re * x ;
        }else {
            result = re * re ;
        }
        return result;
    }
}
